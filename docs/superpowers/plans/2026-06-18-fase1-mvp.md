```typescript
import { db } from '../lib/db.js';
import { logger } from '../lib/logger.js';
import { encryptToken, decryptToken } from '../lib/crypto.js';
import { env } from '../lib/env.js';

const ML_BASE = 'https://api.mercadolibre.com';
const ML_AUTH_URL = 'https://auth.mercadolivre.com.br/authorization';

// ── Token management ──────────────────────────────────────────────────────────

async function getAccessToken(sellerId: string): Promise<string> {
  const { rows } = await db.query<{
    access_token: string;
    refresh_token: string;
    expires_at: Date;
  }>(
    'SELECT access_token, refresh_token, expires_at FROM ml_credentials WHERE seller_id = $1',
    [sellerId]
  );

  if (!rows[0]) throw new Error(`Credenciais não encontradas para seller ${sellerId}`);

  const expiresAt = new Date(rows[0].expires_at);
  const oneHour = 60 * 60 * 1000;

  if (expiresAt.getTime() - Date.now() < oneHour) {
    return refreshToken(sellerId, decryptToken(rows[0].refresh_token));
  }

  return decryptToken(rows[0].access_token);
}

async function refreshToken(sellerId: string, refreshToken: string): Promise<string> {
  const resp = await fetch(`${ML_BASE}/oauth/token`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({
      grant_type: 'refresh_token',
      client_id: env.ML_APP_ID,
      client_secret: env.ML_APP_SECRET,
      refresh_token: refreshToken,
    }),
  });

  if (!resp.ok) {
    const body = await resp.text();
    throw new Error(`Token refresh falhou: ${resp.status} ${body}`);
  }

  const data = await resp.json() as {
    access_token: string;
    refresh_token: string;
    expires_in: number;
  };

  const expiresAt = new Date(Date.now() + data.expires_in * 1000);

  await db.query(
    `UPDATE ml_credentials
       SET access_token = $1, refresh_token = $2, expires_at = $3, updated_at = NOW()
     WHERE seller_id = $4`,
    [encryptToken(data.access_token), encryptToken(data.refresh_token), expiresAt, sellerId]
  );

  logger.info({ sellerId }, 'Access token renovado');
  return data.access_token;
}

// ── OAuth helpers ─────────────────────────────────────────────────────────────

export function getOAuthUrl(): string {
  const params = new URLSearchParams({
    response_type: 'code',
    client_id: env.ML_APP_ID,
    redirect_uri: env.ML_REDIRECT_URI,
  });
  return `${ML_AUTH_URL}?${params}`;
}

export async function exchangeCodeForTokens(code: string): Promise<{
  access_token: string;
  refresh_token: string;
  expires_in: number;
  user_id: number;
}> {
  const resp = await fetch(`${ML_BASE}/oauth/token`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({
      grant_type: 'authorization_code',
      client_id: env.ML_APP_ID,
      client_secret: env.ML_APP_SECRET,
      code,
      redirect_uri: env.ML_REDIRECT_URI,
    }),
  });

  if (!resp.ok) {
    const body = await resp.text();
    throw new Error(`Troca de code falhou: ${resp.status} ${body}`);
  }

  return resp.json();
}

export async function saveCredentials(
  sellerId: string,
  accessToken: string,
  refreshToken: string,
  expiresIn: number
): Promise<void> {
  const expiresAt = new Date(Date.now() + expiresIn * 1000);
  await db.query(
    `INSERT INTO ml_credentials (seller_id, access_token, refresh_token, expires_at)
     VALUES ($1, $2, $3, $4)
     ON CONFLICT (seller_id) DO UPDATE SET
       access_token  = EXCLUDED.access_token,
       refresh_token = EXCLUDED.refresh_token,
       expires_at    = EXCLUDED.expires_at,
       updated_at    = NOW()`,
    [sellerId, encryptToken(accessToken), encryptToken(refreshToken), expiresAt]
  );
}

// ── ML API calls ──────────────────────────────────────────────────────────────

export interface MLQuestion {
  id: number;
  text: string;
  item_id: string;
  seller_id: number;
  status: string;
  date_created: string;
}

export interface MLItem {
  id: string;
  title: string;
  plain_text?: string;
  attributes?: Array<{ id: string; name: string; value_name: string }>;
}

export async function getQuestion(questionId: string, sellerId: string): Promise<MLQuestion> {
  const token = await getAccessToken(sellerId);
  const resp = await fetch(`${ML_BASE}/questions/${questionId}`, {
    headers: { Authorization: `Bearer ${token}` },
  });
  if (!resp.ok) throw new Error(`GET /questions/${questionId} → ${resp.status}`);
  return resp.json();
}

export async function getItem(itemId: string, sellerId: string): Promise<MLItem> {
  const token = await getAccessToken(sellerId);
  const resp = await fetch(`${ML_BASE}/items/${itemId}?include_attributes=all`, {
    headers: { Authorization: `Bearer ${token}` },
  });
  if (!resp.ok) throw new Error(`GET /items/${itemId} → ${resp.status}`);
  return resp.json();
}

export async function postAnswer(
  questionId: string,
  text: string,
  sellerId: string
): Promise<{ id: number; status: string }> {
  const token = await getAccessToken(sellerId);
  const resp = await fetch(`${ML_BASE}/answers`, {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${token}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({ question_id: Number(questionId), text }),
  });
  if (!resp.ok) {
    const body = await resp.text();
    throw new Error(`POST /answers → ${resp.status}: ${body}`);
  }
  return resp.json();
}
```

```typescript
import Groq from 'groq-sdk';
import { env } from '../lib/env.js';
import { logger } from '../lib/logger.js';

const client = new Groq({ apiKey: env.GROQ_API_KEY });

const DEFAULT_SYSTEM_PROMPT = `Você é um atendente especializado em peças automotivas e acessórios de corrida da Expert Racing Parts.
Responda perguntas de compradores no Mercado Livre de forma direta, amigável e profissional.

Regras obrigatórias:
- Máximo 280 caracteres na resposta
- Sempre em português do Brasil
- Vá direto ao ponto — sem saudações longas
- Nunca mencione concorrentes
- Se não tiver certeza, diga "Sim, verificamos nosso estoque e [resposta]" ou peça contato
- Nunca ofereça desconto sem autorização explícita`;

export interface AnswerContext {
  question: string;
  productTitle: string;
  productDescription: string;
  ragContext: string;           // Q&As similares (vazio na Fase 1)
  customInstructions?: string;  // Instruções específicas do produto
  systemPrompt?: string;        // System prompt customizado pelo cliente
}

export async function generateAnswer(ctx: AnswerContext): Promise<string> {
  const systemContent = ctx.systemPrompt || DEFAULT_SYSTEM_PROMPT;

  const ragSection = ctx.ragContext
    ? `\nPerguntas similares respondidas anteriormente:\n${ctx.ragContext}\n`
    : '';

  const customSection = ctx.customInstructions
    ? `\nInstruções especiais para este produto: ${ctx.customInstructions}\n`
    : '';

  const userContent = `Produto: ${ctx.productTitle}
Descrição: ${ctx.productDescription.slice(0, 600)}
${ragSection}${customSection}
Pergunta do comprador: ${ctx.question}

Responda (máximo 280 caracteres):`;

  const start = Date.now();

  const completion = await client.chat.completions.create({
    model: 'llama-3.3-70b-versatile',
    messages: [
      { role: 'system', content: systemContent },
      { role: 'user', content: userContent },
    ],
    temperature: 0.3,
    max_tokens: 200,
  });

  const raw = completion.choices[0]?.message?.content?.trim() ?? '';
  const answer = raw.length > 300 ? raw.slice(0, 297) + '...' : raw;

  logger.debug({
    questionPreview: ctx.question.slice(0, 50),
    answerLength: answer.length,
    groqMs: Date.now() - start,
  }, 'Groq answer generated');

  if (!answer) throw new Error('Groq retornou resposta vazia');

  return answer;
}
```

```typescript
// Payload que o ML envia no webhook de perguntas
export interface MLWebhookPayload {
  resource: string;       // ex: "/questions/5036111111"
  user_id: number;        // seller_id numérico
  topic: string;          // "questions"
  application_id: number;
}

export function validateWebhookPayload(body: unknown): body is MLWebhookPayload {
  if (typeof body !== 'object' || body === null) return false;
  const b = body as Record<string, unknown>;
  return (
    typeof b.resource === 'string' &&
    typeof b.user_id === 'number' &&
    typeof b.topic === 'string' &&
    typeof b.application_id === 'number'
  );
}

export function extractQuestionId(resource: string): string | null {
  // resource: "/questions/5036111111"
  const match = resource.match(/\/questions\/(\d+)/);
  return match ? match[1] : null;
}
```

```typescript
  durationMs?: number;
}

export async function processQuestion(
  questionId: string,
  sellerId: string
): Promise<OrchestrationResult> {
  const start = Date.now();

  try {
    // 1. Buscar pergunta + produto na ML API em paralelo
    const question = await getQuestion(questionId, sellerId);
    const item = await getItem(question.item_id, sellerId);

    const questionText = question.text;
    const itemId = question.item_id;

    // 2. Buscar configurações customizadas para este produto
    const { rows: productRows } = await db.query<{
      custom_instructions: string | null;
      system_prompt: string | null;
    }>(
      `SELECT p.custom_instructions, bs.system_prompt
       FROM products p
       LEFT JOIN bot_settings bs ON bs.seller_id = p.seller_id
       WHERE p.ml_item_id = $1 AND p.seller_id = $2`,
      [itemId, sellerId]
    );

    // Garantir produto no banco (upsert básico)
    await db.query(
      `INSERT INTO products (id, seller_id, ml_item_id, title, description)
       VALUES ($1, $2, $3, $4, $5)
       ON CONFLICT (id) DO UPDATE SET
         title = EXCLUDED.title,
         description = EXCLUDED.description,
         updated_at = NOW()`,
      [itemId, sellerId, itemId, item.title, item.plain_text ?? '']
    );

    const customInstructions = productRows[0]?.custom_instructions ?? undefined;
    const systemPrompt = productRows[0]?.system_prompt ?? undefined;

    // 3. Gerar resposta com Groq (RAG vazio na Fase 1 — implementado na Fase 2)
    const answer = await generateAnswer({
      question: questionText,
      productTitle: item.title,
      productDescription: item.plain_text ?? item.title,
      ragContext: '',  // Fase 2: preencher com pgvector search
      customInstructions,
      systemPrompt,
    });

    // 4. Postar resposta no ML
    await postAnswer(questionId, answer, sellerId);

    // 5. Salvar no histórico (embedding null — Fase 2 adiciona)
    await db.query(
      `INSERT INTO qa_history (product_id, ml_question_id, question, answer, status)
       VALUES ($1, $2, $3, $4, 'success')
       ON CONFLICT (ml_question_id) DO NOTHING`,
      [itemId, questionId, questionText, answer]
    );

    const durationMs = Date.now() - start;
    logger.info({ questionId, sellerId, durationMs, answerPreview: answer.slice(0, 60) },
      'Pergunta respondida com sucesso');

    return { success: true, answer, durationMs };

  } catch (error) {
    const message = error instanceof Error ? error.message : String(error);
    const durationMs = Date.now() - start;
    logger.error({ questionId, sellerId, error: message, durationMs }, 'Falha ao processar pergunta');
    return { success: false, error: message, durationMs };
  }
}
```

```typescript
import type { FastifyInstance } from 'fastify';
import { validateWebhookPayload, extractQuestionId } from './validator.js';
import { processQuestion } from '../bot/orchestrator.js';
import { db } from '../lib/db.js';
import { logger } from '../lib/logger.js';

export async function webhookRoutes(app: FastifyInstance): Promise<void> {
  // Health check
  app.get('/health', async () => ({
    ok: true,
    uptime: Math.round(process.uptime()),
    timestamp: new Date().toISOString(),
  }));

  // Webhook principal — ML chama este endpoint para cada pergunta nova
  app.post<{ Body: unknown }>('/webhook/questions', async (request, reply) => {
    const body = request.body;

    // 1. Validar payload
    if (!validateWebhookPayload(body)) {
      logger.warn({ body }, 'Webhook payload inválido');
      return reply.status(400).send({ error: 'Invalid payload' });
    }

    // 2. Ignorar topics que não são perguntas
    if (body.topic !== 'questions') {
      return reply.status(200).send({ ok: true, skipped: true });
    }

    const questionId = extractQuestionId(body.resource);
    const sellerId = String(body.user_id);

    if (!questionId) {
      logger.warn({ resource: body.resource }, 'Não foi possível extrair question ID');
      return reply.status(200).send({ ok: true });
    }

    // 3. Registrar webhook recebido
    const webhookRef = `${body.application_id}-${questionId}`;
    const { rows: logRows } = await db.query(
      `INSERT INTO webhook_logs (webhook_ref, seller_id, question_id, status)
       VALUES ($1, $2, $3, 'processing')
       ON CONFLICT DO NOTHING
       RETURNING id`,
      [webhookRef, sellerId, questionId]
    );

    // Pergunta duplicada (ML pode reenviar) — ignorar
    if (!logRows[0]) {
      logger.debug({ questionId }, 'Webhook duplicado ignorado');
      return reply.status(200).send({ ok: true, duplicate: true });
    }

    const logId = logRows[0].id;

    // 4. Responder 200 para o ML IMEDIATAMENTE (ML exige resposta em <500ms)
    reply.status(200).send({ ok: true });

    // 5. Processar de forma assíncrona (não bloqueia o reply)
    const result = await processQuestion(questionId, sellerId);

    // 6. Atualizar log com resultado
    await db.query(
      `UPDATE webhook_logs
         SET status = $1, error_message = $2
       WHERE id = $3`,
      [result.success ? 'success' : 'failed', result.error ?? null, logId]
    );
  });
}
```

```typescript
  const code = req.nextUrl.searchParams.get('code');
  const error = req.nextUrl.searchParams.get('error');

  if (error || !code) {
    return NextResponse.redirect(new URL(`/connect?error=${error ?? 'no_code'}`, req.url));
  }

  try {
    // Trocar code por tokens
    const tokenResp = await fetch('https://api.mercadolibre.com/oauth/token', {

        client_id: process.env.ML_APP_ID!,
        client_secret: process.env.ML_APP_SECRET!,
        code,
        redirect_uri: process.env.ML_REDIRECT_URI!,
      }),
    });

    if (!tokenResp.ok) {
      const body = await tokenResp.text();
      throw new Error(`Troca de token falhou: ${body}`);
    }

    const tokens = await tokenResp.json() as {
      access_token: string;
      refresh_token: string;
      expires_in: number;
      user_id: number;
    };

    const expiresAt = new Date(Date.now() + tokens.expires_in * 1000);

         expires_at    = EXCLUDED.expires_at,
```

```typescript
import { createCipheriv, createDecipheriv, randomBytes, scryptSync } from 'crypto';
import { env } from './env.js';

// KEY: 32 bytes derivados da ENCRYPTION_KEY (hex de 64 chars)
const KEY = Buffer.from(env.ENCRYPTION_KEY, 'hex');

export function encryptToken(plaintext: string): string {
  const iv = randomBytes(16);
  const cipher = createCipheriv('aes-256-cbc', KEY, iv);
  const encrypted = Buffer.concat([
    cipher.update(plaintext, 'utf8'),
    cipher.final(),
  ]);
  // Formato: iv_hex:encrypted_hex
  return `${iv.toString('hex')}:${encrypted.toString('hex')}`;
}

export function decryptToken(stored: string): string {
  const [ivHex, encHex] = stored.split(':');
  if (!ivHex || !encHex) throw new Error('Token formato inválido');
  const iv = Buffer.from(ivHex, 'hex');
  const enc = Buffer.from(encHex, 'hex');
  const decipher = createDecipheriv('aes-256-cbc', KEY, iv);
  return Buffer.concat([decipher.update(enc), decipher.final()]).toString('utf8');
}
```

```typescript
import { Pool } from 'pg';

async function getStats() {
  const db = new Pool({ connectionString: process.env.DATABASE_URL });
  try {
    const [totalRows, successRows, credRows, logsRows] = await Promise.all([
      db.query('SELECT COUNT(*) FROM qa_history'),
      db.query("SELECT COUNT(*) FROM qa_history WHERE status = 'success'"),
      db.query('SELECT seller_id, expires_at FROM ml_credentials LIMIT 1'),
      db.query(`SELECT status, COUNT(*) FROM webhook_logs
                WHERE created_at > NOW() - INTERVAL '24h'
                GROUP BY status`),
    ]);
    return {
      total: Number(totalRows.rows[0].count),
      success: Number(successRows.rows[0].count),
      connected: credRows.rows[0]?.seller_id ?? null,
      expiresAt: credRows.rows[0]?.expires_at ?? null,
      logs24h: logsRows.rows as Array<{ status: string; count: string }>,
    };
  } finally {
    await db.end();
  }
}

          { label: '// total respondidas', value: stats.total, color: 'var(--navy)' },
```
